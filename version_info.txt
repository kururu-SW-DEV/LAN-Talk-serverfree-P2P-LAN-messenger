# PyInstaller Windows 버전 리소스 — exe에 이 정보를 심어두면 Windows가 여러 UI
# (토스트 알림, 파일 속성, Alt+Tab 도구 설명 등)에서 원본 파일명("LAN Talk.exe")
# 대신 여기 적힌 FileDescription/ProductName("LAN Talk")을 대신 보여준다.
# 빌드 시 `--version-file=version_info.txt`로 사용한다.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(6, 48, 0, 0),
    prodvers=(6, 48, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',
          [StringStruct('CompanyName', 'LAN Talk'),
          StringStruct('FileDescription', 'LAN Talk'),
          StringStruct('FileVersion', '6.48.0.0'),
          StringStruct('InternalName', 'LAN Talk'),
          StringStruct('OriginalFilename', 'LAN Talk.exe'),
          StringStruct('ProductName', 'LAN Talk'),
          StringStruct('ProductVersion', '6.48.0.0')])
      ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
